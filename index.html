// ============================================================ // Google Apps Script — Code.gs // Sheet1 : بيانات التسجيل // Sheet2 : الإحصائيات والتسليم // ============================================================ const REG_SHEET = 'Sheet1'; const STATS_SHEET = 'Sheet2'; // ──────────────────────────────────────────── // Helpers // ──────────────────────────────────────────── function normalizeText(value) { return String(value || '').trim().replace(/\s+/g, ' ').toLowerCase(); } function getDocColumnIndex(docType) { if (docType === 'دفتر العائلة') return 6; // G if (docType === 'البيان العائلي') return 7; // H if (docType === 'مطلقة') return 8; // I return -1; } function mapRow(r) { return { refNum: r[0], timestamp: r[1], regType: r[2], registrationPurpose: r[3], gasAgent: r[4], docType: r[5], familyBookNum: r[6], stmtDocNum: r[7], divDocNum: r[8], qaidNum: r[9], stampStatus: r[10], personName: r[11], familyCount: r[12], familyLocation: r[13], neighborhood: r[14], street: r[15], addressDetail: r[16], fullAddress: r[17], municipality: r[18], status: r[19], receivedAt: r[20] }; } function checkDuplicate(params) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(REG_SHEET); if (!sheet) return { status: 'ERROR', msg: 'ورقة التسجيلات غير موجودة' }; const rows = sheet.getDataRange().getValues(); if (rows.length <= 1) { return { status: 'OK', exists: false, docExists: false, personQaidExists: false, msg: '' }; } const data = rows.slice(1); const personName = normalizeText(params.personName || ''); const qaidNum = String(params.qaidNum || '').trim(); const docType = String(params.docType || '').trim(); const docNum = String(params.docNum || '').trim(); const docCol = getDocColumnIndex(docType); const docExists = (docCol !== -1 && docNum) ? data.some(r => String(r[docCol] || '').trim() === docNum) : false; const personQaidExists = (personName && qaidNum) ? data.some(r => normalizeText(r[11]) === personName && String(r[9] || '').trim() === qaidNum ) : false; let msg = ''; if (docExists && personQaidExists) { msg = 'رقم الوثيقة مسجل مسبقاً، كما أن الاسم الكامل مع رقم القيد مسجلان مسبقاً'; } else if (docExists) { msg = 'رقم الوثيقة مسجل مسبقاً'; } else if (personQaidExists) { msg = 'الاسم الكامل مع رقم القيد مسجلان مسبقاً'; } return { status: 'OK', exists: docExists || personQaidExists, docExists, personQaidExists, msg }; } // ──────────────────────────────────────────── // doPost — استقبال تسجيل جديد // ──────────────────────────────────────────── function doPost(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(REG_SHEET); const data = JSON.parse(e.postData.contents); const rows = sheet.getDataRange().getValues().slice(1); const incomingName = normalizeText(data.personName || data.husbandFullName || ''); const incomingQaid = String(data.qaidNum || '').trim(); const incomingDocType = String(data.docType || '').trim(); const incomingDocNum = incomingDocType === 'دفتر العائلة' ? String(data.familyBookNum || '').trim() : incomingDocType === 'البيان العائلي' ? String(data.stmtDocNum || '').trim() : incomingDocType === 'مطلقة' ? String(data.divDocNum || '').trim() : ''; const docCol = getDocColumnIndex(incomingDocType); const duplicateDoc = (docCol !== -1 && incomingDocNum) ? rows.some(r => String(r[docCol] || '').trim() === incomingDocNum) : false; const duplicatePersonQaid = (incomingName && incomingQaid) ? rows.some(r => normalizeText(r[11]) === incomingName && String(r[9] || '').trim() === incomingQaid ) : false; if (duplicateDoc || duplicatePersonQaid) { return json({ status: 'ERROR', msg: duplicateDoc && duplicatePersonQaid ? 'رقم الوثيقة مسجل مسبقاً، كما أن الاسم الكامل مع رقم القيد مسجلان مسبقاً' : duplicateDoc ? 'رقم الوثيقة مسجل مسبقاً' : 'الاسم الكامل مع رقم القيد مسجلان مسبقاً' }); } sheet.appendRow([ data.refNum || '', data.timestamp || '', data.regType || '', data.registrationPurpose || '', data.gasAgent || '', data.docType || '', data.familyBookNum || '', data.stmtDocNum || '', data.divDocNum || '', data.qaidNum || '', data.stampStatus || '', data.personName || data.husbandFullName || '', data.familyCount || '', data.familyLocation || '', data.neighborhood || '', data.street || '', data.addressDetail || '', data.fullAddress || '', data.municipality || 'بلدية داعل — محافظة درعا', 'قيد المراجعة', new Date().toLocaleString('ar-SY') ]); return json({ status: 'OK', ref: data.refNum }); } catch(err) { return json({ status: 'ERROR', msg: err.message }); } } // ──────────────────────────────────────────── // doGet // ──────────────────────────────────────────── function doGet(e) { try { const action = (e.parameter.action || '').toLowerCase(); if (action === 'stats') return json(getStats()); if (action === 'search') return json(doSearch(e.parameter)); if (action === 'check_duplicate') return json(checkDuplicate(e.parameter)); return json({ status: 'ERROR', msg: 'action مطلوب: stats أو search أو check_duplicate' }); } catch(err) { return json({ status: 'ERROR', msg: err.message }); } } // ──────────────────────────────────────────── // getStats // ──────────────────────────────────────────── function getStats() { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(STATS_SHEET); if (!sheet) { return { status: 'ERROR', msg: 'Sheet2 غير موجودة — تأكد من اسم الورقة' }; } const rows = sheet.getDataRange().getValues(); const data = {}; rows.forEach(row => { const key = String(row[0]).trim(); const val = row[1]; if (key && key !== 'المفتاح') data[key] = val; }); return { status: 'OK', lastDelivered: data['last_delivered'] || '—', rangeFrom: data['range_from'] || '—', rangeTo: data['range_to'] || '—', weekLabel: data['week_label'] || '—', total: Number(data['total']) || 0, delivered: Number(data['delivered']) || 0, unstamped: Number(data['unstamped']) || 0, books: Number(data['books']) || 0, stmts: Number(data['stmts']) || 0, divs: Number(data['divs']) || 0, lastUpdated: data['last_updated'] || new Date().toLocaleString('ar-SY'), }; } // ──────────────────────────────────────────── // doSearch // ──────────────────────────────────────────── function doSearch(params) { const by = (params.by || '').toLowerCase(); const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(REG_SHEET); if (!sheet) return { status: 'ERROR', msg: 'ورقة التسجيلات غير موجودة' }; const rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return { status: 'OK', result: null }; const data = rows.slice(1); if (by === 'secure_doc') { const ref = (params.ref || '').trim(); const doc = (params.doc || '').trim(); if (!ref) return { status: 'ERROR', msg: 'الرقم المرجعي مطلوب' }; if (!doc) return { status: 'ERROR', msg: 'رقم الوثيقة مطلوب' }; const match = data.find(r => { const refMatch = normalizeText(r[0]) === normalizeText(ref); if (!refMatch) return false; const docMatch = [6, 7, 8, 9].some(i => { const val = String(r[i] || '').trim(); return val !== '' && val !== '—' && val === doc; }); return docMatch; }); return { status: 'OK', result: match ? mapRow(match) : null }; } if (by === 'secure_name') { const ref = (params.ref || '').trim(); const name = (params.name || '').trim(); if (!ref) return { status: 'ERROR', msg: 'الرقم المرجعي مطلوب' }; if (!name) return { status: 'ERROR', msg: 'الاسم الكامل مطلوب' }; const match = data.find(r => { const refMatch = normalizeText(r[0]) === normalizeText(ref); const nameMatch = normalizeText(r[11]) === normalizeText(name); return refMatch && nameMatch; }); return { status: 'OK', result: match ? mapRow(match) : null }; } return { status: 'ERROR', msg: 'نوع البحث غير صحيح' }; } // ──────────────────────────────────────────── // json helper // ──────────────────────────────────────────── function json(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON);
